```ts
// chrono reads "this week" as its first day (last Sunday), certain; it means the week it was said in.
const THIS_WEEK = /\b(later )?this week\b/i;
const WEEK_END = /\bend of (the|this) week\b/i;
const WEEKDAYS = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"];

/** Monday of the week `ref` falls in, in their zone. */
function mondayOf(ref: Date, zone: string): Date {
  const back = WEEKDAYS.indexOf(parts(ref, zone).weekday!);
  return new Date(ref.getTime() - back * DAY);
}
```

```ts
const HERS: ReadonlySet<CandidateKind> = new Set(["her_day", "core_thought"]);

/**
 * The week's last slot is kept for them: with one left, her own news waits, so a thread that
 * comes due later in the week still has room.
 */
export function holdLastSlot(
  candidates: Candidate[],
  sentThisWeek: number,
  settings: OutreachConfig,
): Candidate[] {
  if (settings.cap_per_week - sentThisWeek > 1) return candidates;
  return candidates.map((c) =>
    HERS.has(c.kind) && c.filtered === null ? { ...c, filtered: "last_slot" } : c,
  );
}
```

```ts
/**
 * A quiet block holds the reply until it ends: she's offline, so a reply in the middle of it would
 * be a reply from nowhere. A busy block delays by busy × (1 − closeness / override) × max delay,
 * jittered by `unit` (0–1), capped at the block's end. Closeness is the param's value, not its
 * deviation. At or above `override_closeness` there's no delay.
 */
export function availabilityDelay(
  block: Pick<WorldBlock, "busy" | "block" | "end">,
  closeness: number,
  now: Date,
  unit: number,
  settings: AvailabilityConfig,
): number {
  if (closeness >= settings.override_closeness) return 0;
  const untilBlockEnds = Math.max(0, block.end.getTime() - now.getTime());
  if (block.block === "quiet") return untilBlockEnds;

  const distance = 1 - closeness / settings.override_closeness;
  const jitter = 1 - settings.jitter + 2 * settings.jitter * unit;
  const delay = block.busy * distance * settings.max_delay_ms * jitter;
  return Math.max(0, Math.round(Math.min(delay, untilBlockEnds)));
}
```

```ts
export function simUserInput(
  card: string,
  turns: { role: "user" | "companion"; text: string; at: Date }[],
  now: Date,

  const convo = turns
```
