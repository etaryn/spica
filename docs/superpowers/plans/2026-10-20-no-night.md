# No Night Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remove Spica's night cycle.
- Her energy becomes a battery: busy blocks drain it, free time restores it.
- When it runs low she logs off, at any hour, for as long as it takes to recover.
- Offline acts like a busy block wherever her night did.
- Reach-outs respect the user's own night.
- Her plan, browsing and posting run at any hour.

**Architecture:**
- **Core state:** `core_state_tick` computes energy as a running battery and owns the offline
  switch, stored in a new column `core_state.offline_since` with hysteresis between two
  thresholds.
- **Blocks:** `todaysBlocks` returns her busy blocks plus one synthetic offline block while
  she's offline. Every existing reader (availability, outreach G3, wander, posts, "Your day",
  openness) gets offline for free.
- **Plans:** day plans lose the quiet kind entirely.
- **Outreach:** gains a `their_night` gate in the user's timezone.

**Tech Stack:** TypeScript, Drizzle + Postgres (one migration), Vitest, pnpm.

**Spec:**
- `spica-design-spec-v3.md` §8.3 (world), §8.4 (core state), §10.6 (availability), §13
  (outreach), §16a (posts).
- Design doc, settled lines: `docs/2026-10-20-no-night-design.md`.
- Memory: `no-night-settled`.

## Review Focus

- **Offline that starts between ticks.** A message arriving at :50 after the :45 tick set
  offline: `todaysBlocks` must see it immediately (it reads the row), and the block's estimated
  end must be in the future. A past end gets clamped to `last_tick + 1 h`. (Test in Task 2.)
- **An old stored plan containing `quiet` blocks** (dev DB, before this change). It must read as
  free time, never throw, and never produce a block. (Test in Task 2.)
- **A user whose timezone is null or invalid.** The `their_night` gate uses UTC and never throws.
  (Test in Task 5.)
- **`offline.enter >= offline.exit` in config.** Config validation must reject it, or she would
  flicker on and off every tick. (Test in Task 1.)
- **A deferral stored with an old `<date>#<HH:MM>` key for a block that was quiet.**
  `blockByKey` returns null (no "while you were" line) rather than failing. An `offline#…` key
  returns her offline label. (Test in Task 2.)

```ts
/** The busy-weighted and plain fractions of [from, to) she spent in busy blocks. */
export function hourLoad(
  blocks: WorldBlock[],
  from: Date,
  to: Date,
): { load: number; covered: number } {
  const span = to.getTime() - from.getTime();
  let load = 0;
  let covered = 0;
  for (const b of blocks) {
    if (b.block !== "busy") continue;
    const overlap =
      Math.min(b.end.getTime(), to.getTime()) - Math.max(b.start.getTime(), from.getTime());
    if (overlap > 0) {
      load += (b.busy * overlap) / span;
      covered += overlap / span;
    }
  }
  return { load: clamp(load, 0, 1), covered: clamp(covered, 0, 1) };
}

/**
 * One hour of her battery: busy time drains it, free time slowly restores it, and offline restores
 * it fast. Offline, her plan doesn't happen, so its blocks don't count.
 */
export function energyStep(
  energy: number,
  hour: { load: number; covered: number },
  offline: boolean,
  noise: number,
  cfg: CoreStateConfig,
): number {
  const e = cfg.energy;
  const delta = offline
    ? e.offline_recovery_per_hour
    : -e.busy_drain_per_hour * hour.load + e.free_recovery_per_hour * (1 - hour.covered);
  return clamp(energy + delta + noise * cfg.noise, 0, 1);
}

/** Logs off below `enter`, comes back at `exit`; between them she stays as she was. */
export function nextOffline(
  offlineSince: Date | null,
  energy: number,
  now: Date,
  cfg: CoreStateConfig,
): Date | null {
  if (!offlineSince) return energy < cfg.offline.enter ? now : null;
  return energy >= cfg.offline.exit ? null : offlineSince;
}

/** When she should be back: the hour her energy reaches `exit`, never before the next tick. */
export function estimatedBack(
  row: { energy: number; last_tick: Date | null; offline_since: Date | null },
  cfg: CoreStateConfig,
): Date {
  const from = (row.last_tick ?? row.offline_since ?? new Date(0)).getTime();
  const hours = Math.ceil(
    Math.max(0, cfg.offline.exit - row.energy) / cfg.energy.offline_recovery_per_hour,
  );
  return new Date(from + Math.max(1, hours) * HOUR);
}
```

**`blocks.ts`** (full rewrite of the reading half):

```ts
/**
 * Her busy blocks, read straight from the shared day plan, plus her offline time when her energy
 * has run out. Downtime is free time, so it isn't a block. A plan block's key is her local date and
 * start time; an offline block's is `offline#<when she logged off>`.
 */
import type { LifePlan } from "@/db/schema";
import type { LifeConfig } from "@/mind/platform/config";
import type { Deps } from "@/mind/platform/deps";
import { estimatedBack } from "@/mind/state/core/core-state";
import { lifeDate, lifeInstant } from "./her-clock";

export type WorldBlock = {
  key: string;
  start: Date;
  end: Date;
  block: "busy" | "offline";
  busy: number;
  label: string;
};

const KEY = /^(\d{4}-\d{2}-\d{2})#(\d{2}:\d{2})$/;
const OFFLINE = /^offline#(.+)$/;

export function blocksForDay(
  settings: LifeConfig,
  day: string,
  plan: LifePlan | null,
): WorldBlock[] {
  if (!plan) return [];
  return plan.blocks
    .filter((b) => b.kind === "busy")
    .map((b) => ({
      key: `${day}#${b.start}`,
      start: lifeInstant(settings, day, b.start),
      end: lifeInstant(settings, day, b.end),
      block: "busy" as const,
      busy: b.busy,
      label: b.label,
    }))
    .sort((a, b) => a.start.getTime() - b.start.getTime());
}

export const blockAt = (blocks: WorldBlock[], at: Date): WorldBlock | null =>
  blocks.find((b) => b.start.getTime() <= at.getTime() && at.getTime() < b.end.getTime()) ?? null;

/** Her blocks now: offline first while she's logged off (it wins over her plan), then the plan's. */
export async function todaysBlocks(
  deps: Pick<Deps, "core" | "config">,
  now: Date,
): Promise<WorldBlock[]> {
  const day = lifeDate(deps.config.life, now);
  const [plan, state] = await Promise.all([deps.core.world.plan(day), deps.core.state.get()]);
  const planned = blocksForDay(deps.config.life, day, plan);
  if (!state?.offline_since) return planned;
  const cfg = deps.config.core.state;
  const offline: WorldBlock = {
    key: `offline#${state.offline_since.toISOString()}`,
    start: state.offline_since,
    end: estimatedBack(state, cfg),
    block: "offline",
    busy: cfg.offline.busy,
    label: cfg.offline.label,
  };
  return [offline, ...planned];
}

export async function blockByKey(
  deps: Pick<Deps, "core" | "config">,
  key: string,
): Promise<WorldBlock | null> {
  const off = OFFLINE.exec(key);
  if (off) {
    const start = new Date(off[1]!);
    if (Number.isNaN(start.getTime())) return null;
```

```ts
test("1: a busy stretch drains her, she logs off at any hour, and offline holds a reply like busy", async () => {
  const d = makeTestDeps({ core: { state: { noise: 0 } }, life: { timezone: "UTC" } });
  await busyAllDay(d, "2026-01-05");
  const core = coreDepsOf(d);
  let off: Date | null = null;
  for (let h = 1; h <= 23 && !off; h++) {
    await coreStateTick(core, hour(h));
    off = (await d.core.state.get())!.offline_since;
  }
  expect(off).not.toBeNull();
  const u = await createTestUser();
  await ensureRelationship(d.db, u, hour(0), d.config.relationship);
  const msgAt = new Date(off!.getTime() + 5 * 60_000);
  const a = await deferWhileBusy(d, u, msgAt, "seed");
  expect(a.delayMs).toBeGreaterThan(0);
  expect(a.blockKey).toMatch(/^offline#/);
  expect(a.blockKind).toBe("offline");
  expect(await loadYourDayLine(d, msgAt, null)).toContain(d.config.core.state.offline.label);
});

test("2: she comes back when her battery reaches exit; her plan doesn't drain her while offline", async () => {
  const d = makeTestDeps({ core: { state: { noise: 0 } }, life: { timezone: "UTC" } });
  await busyAllDay(d, "2026-01-05");
  await busyAllDay(d, "2026-01-06");
  const core = coreDepsOf(d);
  let h = 1;
  while (!(await d.core.state.get())?.offline_since && h < 24) await coreStateTick(core, hour(h++));
  const low = (await d.core.state.get())!.energy;
  await coreStateTick(core, hour(h++));
  expect((await d.core.state.get())!.energy).toBeGreaterThan(low); // busy plan, yet recovering
  while ((await d.core.state.get())!.offline_since && h < 48) await coreStateTick(core, hour(h++));
  expect((await d.core.state.get())!.offline_since).toBeNull();
  expect((await d.core.state.get())!.energy).toBeGreaterThanOrEqual(
    d.config.core.state.offline.exit,
  );
});

test("3: a reach-out never goes out in their night", async () => {

const CORE_TABLES = ["core_memory", "core_state", "core_goal", "core_snapshot", "world_event"];
async function coreDump(): Promise<string> {
  const parts: string[] = [];
  for (const t of CORE_TABLES) {
    const res = await getTestDb().execute(
      sql.raw(`SELECT row_to_json(x)::text AS j FROM "${t}" x ORDER BY 1`),
    );
    parts.push(...res.rows.map((r) => String(r.j)));

  return parts.join("\n");
}

test("5: leak canary and side channel: A's data never reaches B's reach-out; reach-outs leave core (energy and offline included) unchanged", async () => {
  const at = new Date("2026-01-05T15:25:00Z");
  const d = makeTestDeps({ life: { timezone: "UTC" }, core: { world: { events_per_day_max: 0 } } });

  const send = async (u: string) => {

  for (const c of await send(b)) expect(JSON.stringify(c)).not.toContain(TOKEN);

  expect(await coreDump()).toBe(before);
```
